cmake_minimum_required(VERSION 3.15)
set(PICO_BOARD pico)
set(PICO_PLATFORM rp2040)
include(pico_sdk_import.cmake)
project(134_led_module)
pico_sdk_init()
add_executable(${PROJECT_NAME} main.c
    led/led.c
    logging/log.c
    device/device.c
)
target_link_libraries(${PROJECT_NAME} pico_stdlib pico_unique_id)
target_include_directories(${PROJECT_NAME} PRIVATE led logging device)
target_compile_definitions(${PROJECT_NAME} PRIVATE DEVICE_BOARD="${PICO_BOARD}")
if(DEFINED LOG_LEVEL)
    target_compile_definitions(${PROJECT_NAME} PRIVATE LOG_LEVEL=${LOG_LEVEL})
endif()
pico_set_linker_script(${PROJECT_NAME} ${CMAKE_CURRENT_SOURCE_DIR}/memmap_rp2040.ld)
pico_enable_stdio_usb(${PROJECT_NAME} 1)
pico_enable_stdio_uart(${PROJECT_NAME} 0)
pico_add_extra_outputs(${PROJECT_NAME})
